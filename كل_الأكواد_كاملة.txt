 return JSON.parse(localStorage.getItem('farm45')||
 '{"birds":0,"start":"","feed":0,"cash":0,"logs":[]}');
}
function save(){
 const d=db();
 const num=document.getElementById('num').value.trim();
 const start=document.getElementById('start').value;
 const birds=Number(document.getElementById('birds').value);
 const price=Number(document.getElementById('price').value);
 const duration=Number(document.getElementById('duration').value);
 if(!num||!start||!Number.isInteger(birds)||birds<1||
 !Number.isFinite(price)||price<0||
 !Number.isInteger(duration)||duration<1){
 alert('راجع رقم الدورة والتاريخ والأعداد والأسعار');return;

 supplier:document.getElementById('supplier').value.trim(),
 breed:document.getElementById('breed').value.trim(),duration};

 d.start=start;
 d.logs=d.logs||[];
 d.logs.push({date:start,
 type:'new_cycle',amount:birds,note:'إنشاء الدورة '+num});
 localStorage.setItem('farm45',JSON.stringify(d));
 document.getElementById('msg').textContent='تم حفظ بيانات الدورة.';
 load();
}
function load(){
 const d=db(),c=d.cycle;
 if(!c){document.getElementById('result').textContent='لا توجد بيانات محفوظة.';return;}

document.getElementById('date').value=today();

function save(){
 const date=document.getElementById('date').value;
 const count=Number(document.getElementById('count').value);
 const total=Number(document.getElementById('total').value);
 const targetText=document.getElementById('target').value;
 const target=targetText===''?null:Number(targetText);
 if(!date||!Number.isInteger(count)||count<1||
 !Number.isFinite(total)||total<=0||
 (target!==null&&(!Number.isFinite(target)||target<=0))){
  alert('راجع التاريخ وعدد الطيور وإجمالي الأوزان');return;
 }
 if(total/count>10){
  alert('متوسط الوزن مرتفع بصورة غير معتادة. راجع وحدة الوزن والقيم.');
  return;
 }
 const d=getDB();
 d.weights=d.weights||[];
 const avg=total/count;
 const previous=d.weights.filter(x=>x.date<date)
  .sort((a,b)=>b.date.localeCompare(a.date))[0];
 const entry={id:Date.now(),date,count,total,
 avg,target,change:previous?avg-previous.avg:null};
 d.weights.push(entry);
 d.logs=d.logs||[];
 d.logs.push({date:date, type:'weight',amount:avg,note:'متوسط وزن العينة'});
 putDB(d);
 document.getElementById('count').value='';
 document.getElementById('total').value='';
 document.getElementById('target').value='';
 document.getElementById('msg').textContent='تم حفظ الأوزان.';
 render();
}
function render(){
 const d=getDB(),arr=(d.weights||[]).slice()
 .sort((a,b)=>b.date.localeCompare(a.date));
 const latest=document.getElementById('latest');
 const records=document.getElementById('records');
 records.replaceChildren();
 if(!arr.length){
  latest.textContent='لا توجد أوزان مسجلة.';
  return;
 }
 const x=arr[0];
 latest.textContent='التاريخ: '+x.date+
 ' | متوسط الطائر: '+x.avg.toFixed(3)+' كجم'+
 (x.target!==null?' | المستهدف: '+x.target+' كجم':'')+
 (x.change!==null?' | التغير عن آخر قياس أقدم: '+
 (x.change>=0?'+':'')+x.change.toFixed(3)+' كجم':'');
 arr.forEach(r=>{
  const p=document.createElement('p');
  p.style.borderBottom='1px solid #ddd';
  p.style.padding='8px 0';
  p.textContent=r.date+' | العينة: '+r.count+

function saveMovement(){
  d = getData();

  const type = document.getElementById("type").value;
  const amount = Number(document.getElementById("amount").value);
  const date = document.getElementById("date").value;
  const note = document.getElementById("note").value.trim();
  const category = document.getElementById("category").value.trim();

  if(!date){
    alert("اختر التاريخ.");
    return;
  }

  if(!Number.isFinite(amount) || amount<=0){
    alert("اكتب مبلغًا صحيحًا أكبر من صفر.");
    return;
  }

  d.cash = Number(d.cash)||0;
  d.cashLogs = d.cashLogs||[];
  d.logs = d.logs||[];

  if(type==="income"){
    if(!category){
      alert("اكتب مصدر الإيراد.");
      return;
    }

    d.cash += amount;
    d.incomes = d.incomes||[];
    d.incomes.push({date,category,amount,note});

    d.cashLogs.unshift({
      date,type:"income",category,amount,note
    });

  }else if(type==="expense"){
    if(!category){
      alert("اكتب بند المصروف.");
      return;
    }

    d.cash -= amount;
    d.expenses = d.expenses||[];
    d.expenses.push({date,category,amount,note});

    d.cashLogs.unshift({
      date,type:"expense",category,amount,note
    });

  }else if(type==="credit"){

    const p = (d.feedPurchases||[])[Number(index)];

    if(index==="" || !p || Number(p.due)<=0){
      alert("اختر فاتورة آجلة مستحقة.");
      return;

    const due = Math.max(0,Number(p.due)||0);

    if(amount>due+0.001){
      alert("مبلغ السداد أكبر من الدين المتبقي: "+money(due));
      return;

      if(!confirm("الرصيد الحالي لا يكفي. هل تريد تسجيل السداد وجعل الرصيد سالبًا؟")){
        return;
      }
    }

    d.cash -= amount;
    p.paid = (Number(p.paid)||0)+amount;
    p.due = Math.max(0,due-amount);

    d.cashLogs.unshift({
      date,
      type:"credit",
      category:"سداد دين علف",
      amount,
      note:note+" | "+(p.supplier||"مورد غير محدد")
    });
  }

  saveData();

  document.getElementById("amount").value="";
  document.getElementById("category").value="";
  document.getElementById("note").value="";

  refresh();
  alert("تم حفظ الحركة.");
}

const KEY="farm45";

function getData(){
  try{
    return JSON.parse(localStorage.getItem(KEY)) ||
      {birds:0,start:"",feed:0,cash:0,logs:[]};
  }catch(e){
    alert("تعذر قراءة بيانات التطبيق.");
    return {birds:0,start:"",feed:0,cash:0,logs:[]};
  }
}

let d=getData();
d.tools=d.tools||[];
d.toolMoves=d.toolMoves||[];

function today(){
  let x=new Date();
  return x.getFullYear()+"-"+String(x.getMonth()+1).padStart(2,"0")+"-"+String(x.getDate()).padStart(2,"0");
}

document.getElementById("date").value=today();
document.getElementById("useDate").value=today();

function money(n){
  return (Number(n)||0).toFixed(2)+" جنيه";
}

function addItem(){
  d= getData();
  d.tools=d.tools||[];
  d.toolMoves=d.toolMoves||[];
  d.logs=d.logs||[];

  let name=document.getElementById("name").value.trim();
  let kind=document.getElementById("kind").value;
  let qty=Number(document.getElementById("qty").value);
  let unit=document.getElementById("unit").value;
  let price=Number(document.getElementById("price").value);
  let date=document.getElementById("date").value;
  let payment=document.getElementById("payment").value;
  let paid=Number(document.getElementById("paid").value);
  let note=document.getElementById("note").value.trim();

  if(!name || !date || !Number.isFinite(qty) || qty<=0 ||
     !Number.isFinite(price) || price<0 ||
     !Number.isFinite(paid) || paid<0){
    alert("راجع اسم الصنف والكمية والسعر والتاريخ.");
    return;
  }

  let total=qty*price;

  if(paid>total+0.001){
    alert("المبلغ المدفوع أكبر من قيمة الشراء.");
    return;
  }

  if(payment==="كاش" && Math.abs(paid-total)>0.001){
    alert("في حالة الكاش يجب أن يساوي المدفوع قيمة الفاتورة.");
    return;
  }

  if(paid>(Number(d.cash)||0)){
    if(!confirm("الرصيد النقدي لا يكفي. هل تريد تسجيل الشراء وجعل الخزنة سالبة؟")){
      return;
    }
  }

  let item=d.tools.find(x=>x.name.toLowerCase()===name.toLowerCase() && x.unit===unit);

  if(item){
    item.qty=(Number(item.qty)||0)+qty;
    item.cost=price;
    item.kind=kind;
  }else{
    item={
      id:Date.now(),
      name:name,
      kind:kind,
      qty:qty,
      unit:unit,
      cost:price
    };
    d.tools.push(item);
  }

  d.cash=(Number(d.cash)||0)-paid;
  d.cashLogs=d.cashLogs||[];
  d.cashLogs.unshift({
    date:date,
    type:"expense",
    category:"شراء أدوات ومستهلكات: "+name,
    amount:paid,
    note:note
  });

  d.toolMoves.unshift({
    date:date,
    type:"شراء",
    name:name,
    qty:qty,
    unit:unit,
    price:price,
    total:total,
    paid:paid,
    due:total-paid,
    payment:payment,
    note:note
  });

  d.logs.push({
    date:date,
    type:"tool_purchase",
    name:name,
    qty:qty,
    total:total,
    paid:paid
  });

  save();
  clearPurchaseForm();
  render();
  alert("تم حفظ الصنف وتحديث المخزن والخزنة.");
}

function useItem(){
  d=getData();
  d.tools=d.tools||[];
  d.toolMoves=d.toolMoves||[];

  let id=Number(document.getElementById("useItem").value);
  let item=d.tools.find(x=>x.id===id);
  let qty=Number(document.getElementById("useQty").value);
  let date=document.getElementById("useDate").value;
  let note=document.getElementById("useNote").value.trim();

  if(!item){
    alert("اختر صنفًا.");
    return;
  }

  if(!Number.isFinite(qty)||qty<=0||!date){
    alert("اكتب كمية وتاريخًا صحيحين.");
    return;
  }

  if(qty>Number(item.qty)){
    alert("الكمية المطلوبة أكبر من الرصيد المتاح.");
    return;
  }

  item.qty=Number(item.qty)-qty;

  d.toolMoves.unshift({
    date:date,
    type:"استهلاك",
    name:item.name,
    qty:qty,
    unit:item.unit,
    total:qty*(Number(item.cost)||0),
    note:note
  });

  save();
  document.getElementById("useQty").value="";
  document.getElementById("useNote").value="";
  render();
  alert("تم تسجيل الاستخدام وتحديث الرصيد.");
}

function save(){
  localStorage.setItem(KEY,JSON.stringify(d));
}

function clearPurchaseForm(){
  ["name","qty","price","note"].forEach(id=>{
    document.getElementById(id).value="";
  });
  document.getElementById("paid").value="0";
}

function render(){
  d=getData();
  d.tools=d.tools||[];
  d.toolMoves=d.toolMoves||[];

  let stock=document.getElementById("stock");
  stock.innerHTML="";

  let select=document.getElementById("useItem");
  select.innerHTML="";

  if(!d.tools.length){
    stock.textContent="لا توجد أصناف.";
  }

  d.tools.forEach(item=>{
    let div=document.createElement("div");
    div.className="item";
    div.textContent=
      item.name+" ("+item.kind+")\n"+
      "الرصيد: "+item.qty+" "+item.unit+"\n"+
      "آخر سعر وحدة: "+money(item.cost);

    if(Number(item.qty)<=1){
      let warning=document.createElement("div");
      warning.className="warn";
      warning.textContent="⚠️ الرصيد منخفض";
      div.appendChild(warning);
    }

    stock.appendChild(div);

    let option=document.createElement("option");
    option.value=item.id;
    option.textContent=item.name+" — المتاح "+item.qty+" "+item.unit;
    select.appendChild(option);
  });

  let history=document.getElementById("history");
  history.innerHTML="";

  if(!d.toolMoves.length){
    history.textContent="لا توجد حركات.";
  }

  d.toolMoves.slice(0,50).forEach(move=>{
    let div=document.createElement("div");
    div.className="item";
    div.textContent=
      (move.date||"")+" — "+move.type+"\n"+
      move.name+" : "+move.qty+" "+(move.unit||"")+"\n"+
      (move.total!==undefined?"القيمة: "+money(move.total)+"\n":"")+
      (move.note||"");
    history.appendChild(div);
  });
}

      <div class="value" id="feed">0 كجم</div>

<div class="card">
  <h3>⚡ إدخال سريع</h3>

  <label>نوع العملية</label>
  <select id="entryType" onchange="changeEntry()">
    <option value="feed">استهلاك علف بالكيلو</option>
    <option value="death">نفوق طيور</option>
    <option value="water">كمية المياه باللتر</option>
    <option value="expense">مصروف نقدي</option>
  </select>

  <label>الكمية أو المبلغ</label>
  <input id="amount" type="number" min="0.01" step="0.01"
         placeholder="أدخل الكمية أو المبلغ">

  <div id="extraField">
    <label>ملاحظات</label>
    <input id="note" placeholder="ملاحظات اختيارية">
  </div>

  <button onclick="saveEntry()">حفظ الإدخال السريع</button>

  <h3>💾 حفظ نسخة احتياطية</h3>
  <p class="small">احفظ نسخة من بيانات المزرعة للاحتفاظ بها.</p>
  <button onclick="backup()">تصدير نسخة احتياطية</button>
</div>

<script>
const KEY="farm45";

function getData(){
  try{
    return JSON.parse(localStorage.getItem(KEY))||
      {birds:0,start:"",feed:0,cash:0,logs:[]};
  }catch(e){
    alert("تعذر قراءة البيانات المحفوظة.");
    return {birds:0,start:"",feed:0,cash:0,logs:[]};
  }
}

function saveData(d){
  localStorage.setItem(KEY,JSON.stringify(d));
}

function today(){const x=new Date();return x.getFullYear()+'-'+String(x.getMonth()+1).padStart(2,'0')+'-'+String(x.getDate()).padStart(2,'0');}

function openSection(){
  const page=document.getElementById("sections").value;
  if(page){
    location.href=page;
  }
}

function changeEntry(){
  const type=document.getElementById("entryType").value;
  const input=document.getElementById("amount");

  if(type==="death"){
    input.placeholder="عدد الطيور النافقة";
    input.step="1";
  }else if(type==="water"){
    input.placeholder="كمية المياه باللتر";
    input.step="0.01";
  }else if(type==="expense"){
    input.placeholder="المبلغ بالجنيه";
    input.step="0.01";
  }else{
    input.placeholder="كمية العلف بالكيلو";
    input.step="0.01";
  }
}

function saveEntry(){
  let d=getData();
  d.logs=d.logs||[];

  const type=document.getElementById("entryType").value;
  const amount=Number(document.getElementById("amount").value);
  const note=document.getElementById("note").value.trim();

  if(!Number.isFinite(amount)||amount<=0){
    alert("أدخل قيمة صحيحة أكبر من صفر.");
    return;
  }

  if(type==="death"&&!Number.isInteger(amount)){
    alert("أدخل عدد الطيور كعدد صحيح.");
    return;
  }

  if(type==="feed"){
    if(amount>Number(d.feed||0)){
      alert("الكمية أكبر من مخزون العلف.");
      return;
    }
    d.feed=Number(d.feed||0)-amount;
  }

  if(type==="death"){
    if(amount>Number(d.birds||0)){
      alert("عدد النفوق أكبر من عدد الطيور المسجل.");
      return;
    }
    d.birds=Number(d.birds||0)-amount;
    d.deaths=d.deaths||[];
    d.deaths.push({
      date:today(),
      count:amount,
      note:note
    });
  }

  if(type==="water"){
    d.waterLogs=d.waterLogs||[];
    d.waterLogs.push({
      date:today(),
      qty:amount,
      note:note
    });
  }

  if(type==="expense"){
    if(amount>Number(d.cash||0)){
      if(!confirm("المبلغ أكبر من رصيد الخزنة. هل تريد المتابعة؟")){
        return;
      }
    }

    d.cash=Number(d.cash||0)-amount;
    d.expenses=d.expenses||[];
    d.expenses.push({
      date:today(),
      category:"مصروف من الإدخال السريع",
      amount:amount,
      note:note
    });

    d.cashLogs=d.cashLogs||[];
    d.cashLogs.unshift({
      date:today(),
      type:"expense",
      category:"مصروف من الإدخال السريع",
      amount:amount,
      note:note
    });
  }

  d.logs.push({
    date:today(),
    type:type,
    qty:amount,
    note:note
  });

  saveData(d);
  document.getElementById("amount").value="";
  document.getElementById("note").value="";
  refresh();
  alert("تم حفظ البيانات.");
}

function backup(){
  const data=localStorage.getItem(KEY)||"{}";
  const blob=new Blob([data],{type:"application/json;charset=utf-8"});
  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");
  a.href=url;
  a.download="backup-mazraat-gedo-sayed.json";
  a.click();
  URL.revokeObjectURL(url);
}

function refresh(){
  const d=getData();

  document.getElementById("birds").textContent=Number(d.birds)||0;
  document.getElementById("feed").textContent=
    (Number(d.feed)||0).toFixed(2)+" كجم";
  document.getElementById("cash").textContent=
    (Number(d.cash)||0).toFixed(2)+" جنيه";

  if(d.start){
    const start=new Date(d.start+"T00:00:00");
    const now=new Date();
    const age=Math.floor((now-start)/86400000)+1;
    document.getElementById("age").textContent=
      age>0?age+" يوم":"-";
  }else{
    document.getElementById("age").textContent="-";
  }
}

refresh();
</script>
